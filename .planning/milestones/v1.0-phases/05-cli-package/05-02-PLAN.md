Per D-07 and research (Kafka is code-generated, not template-copied):

Export `generateKafkaModule(destDir: string, serviceName: string)` that:

1. Creates `src/kafka/` directory in destDir.
2. Writes `src/kafka/kafka.module.ts`:
```typescript
import { Module } from '@nestjs/common';
import { ClientsModule, Transport } from '@nestjs/microservices';
import { ConfigService } from '@nestjs/config';
import { KafkaController } from './kafka.controller';

@Module({
  imports: [
    ClientsModule.registerAsync([{
      name: 'KAFKA_SERVICE',
      useFactory: (configService: ConfigService) => ({
        transport: Transport.KAFKA,
        options: {
          client: {
            brokers: [configService.get<string>('KAFKA_BROKER', 'localhost:9092')],
          },
          consumer: {
            groupId: `${serviceName}-consumer-group`,
          },
        },
      }),
      inject: [ConfigService],
    }]),
  ],
  controllers: [KafkaController],
  exports: [ClientsModule],
})
export class KafkaModule {}
```
